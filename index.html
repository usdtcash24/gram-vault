""", (user_id,)) row = cur.fetchone() conn.close() if not row: tier, mult = calc_tier(user_id) conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute(""" INSERT INTO users (user_id, shards, energy, max_energy, regen_speed, tier, multiplier, has_autoclicker) VALUES (?, 0, 2000, 2000, 2, ?, ?, 0) """, (user_id, tier, mult)) conn.commit() conn.close() return { "user_id": user_id, "shards": 0, "energy": 2000, "max_energy": 2000, "regen_speed": 2, "tier": tier, "multiplier": mult, "has_autoclicker": 0 } return { "user_id": user_id, "shards": row[0], "energy": row[1], "max_energy": row[2], "regen_speed": row[3], "tier": row[4], "multiplier": row[5], "has_autoclicker": row[6] } @app.post("/api/sync") def sync_clicks(data: SyncPayload): conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute( "UPDATE users SET shards = shards + ?, energy = ?, last_sync = CURRENT_TIMESTAMP WHERE user_id = ?", (data.added_shards, data.current_energy, data.user_id) ) conn.commit() conn.close() return {"status": "ok"} @app.post("/api/buy-upgrade") def buy_upgrade(data: UpgradePayload): conn = sqlite3.connect(DB_PATH) cur = conn.cursor() if data.upgrade_type == "battery": cur.execute("UPDATE users SET max_energy = max_energy + 2000 WHERE user_id = ?", (data.user_id,)) elif data.upgrade_type == "charge": cur.execute("UPDATE users SET regen_speed = regen_speed + 4 WHERE user_id = ?", (data.user_id,)) elif data.upgrade_type == "overclock": cur.execute("UPDATE users SET multiplier = multiplier + 2.0 WHERE user_id = ?", (data.user_id,)) elif data.upgrade_type == "autoclicker": cur.execute("UPDATE users SET has_autoclicker = 1 WHERE user_id = ?", (data.user_id,)) else: conn.close() raise HTTPException(status_code=400, detail="Unknown upgrade") conn.commit() conn.close() return {"status": "upgraded", "type": data.upgrade_type} if __name__ == "__main__": port = int(os.environ.get("PORT", 10000)) uvicorn.run("server:app", host="0.0.0.0", port=port)
